\section{项目介绍}

\begin{frame}[allowframebreaks]{项目背景与目标}
    \small
    \textbf{项目名称：}CPME Classification —— 情绪 \& MBTI 推理演示。

    \vspace{0.4em}
    \textbf{要解决的问题：}在中文社交媒体文本上，同时展示两类能力：
    \begin{enumerate}
        \item \textbf{帖子级情绪理解}：给定一条微博风格短文，估计其在六个情绪维度上的强度（连续值 $[0,1]$）。
        \item \textbf{用户级 MBTI 倾向分类}：给定同一用户的多条历史帖子（通常 8--16 条），预测其 MBTI 四字母类型（16 类之一）。
    \end{enumerate}

    \vspace{0.4em}
    \textbf{项目特色（可作 PPT 卖点）：}
    \begin{itemize}
        \item 数据来自公开 CPME 语料，任务定义与论文/课程报告一致，便于复现。
        \item 模型为轻量级 \textbf{字符 CNN / TextCNN}，不依赖 HuggingFace 大模型，\textbf{CPU 即可训练}。
        \item 训练后导出 \textbf{ONNX}，在浏览器通过 \texttt{onnxruntime-web} 完成推理，适合课堂现场演示。
        \item 配套 Web 前端采用分步引导（Landing $\to$ 情绪 $\to$ MBTI $\to$ 总结），界面参考人格类型框架的视觉风格。
    \end{itemize}

    \vspace{0.4em}
    \begin{alertblock}{重要定位（务必在汇报中说明）}
        本系统为\textbf{统计学习研究演示}，输出\textbf{不构成}心理测评、人格诊断、招聘或医疗建议。MBTI 标签来自数据集分层，模型学习的是文本关联模式，而非临床意义上的「人格测量」。
    \end{alertblock}
\end{frame}

\begin{frame}{双模块任务对照表}

    \begin{table}

        \begin{tabular}{p{2cm}p{3.2cm}p{3.2cm}p{4.5cm}}
            \toprule
            & \textbf{输入} & \textbf{输出} & \textbf{样本单位 / 备注} \\
            \midrule
            情绪模块 &
            1 条帖子文本 &
            6 维向量：angry, fear, happy, neutral, sad, surprise &
            以\textbf{单帖}为样本；多标签回归（BCE + sigmoid） \\
            \addlinespace
            MBTI 模块 &
            同一用户的 $P$ 条帖子，$P \leq 16$ &
            16 类 MBTI 类型之一 + 可展示 Top-3 概率 &
            以\textbf{用户}为样本；划分时禁止同一用户跨 train/test \\
            \bottomrule
        \end{tabular}
    \end{table}

\end{frame}
